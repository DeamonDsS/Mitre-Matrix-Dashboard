import type { MitreTechnique, ElasticsearchResponse } from '../types/mitre';

interface EsQueryClause {
  multi_match?: {
    query: string;
    fields: string[];
    fuzziness: string;
  };
  match?: {
    [key: string]: string;
  };
}

// ใช้ proxy server แทนการเรียก ES โดยตรง
const PROXY_URL = 'http://localhost:4000/api/search';

export async function fetchMitreTechniques(
  esUrl: string,
  esIndex: string,
  filters?: {
    search?: string;
    tactic?: string;
    severity?: string;
  }
): Promise<MitreTechnique[]> {
  const query: { bool: { must: EsQueryClause[] } } = {
    bool: {
      must: [],
    },
  };

  if (filters?.search) {
    query.bool.must.push({
      multi_match: {
        query: filters.search,
        fields: ['technique_id', 'technique_name', 'description'],
        fuzziness: 'AUTO',
      },
    });
  }

  if (filters?.tactic && filters.tactic !== 'all') {
    query.bool.must.push({
      match: { tactic: filters.tactic },
    });
  }

  if (filters?.severity && filters.severity !== 'all') {
    query.bool.must.push({
      match: { severity: filters.severity },
    });
  }

  const requestBody = {
    query: query.bool.must.length > 0 ? query : { match_all: {} },
    size: 100,
    sort: [{ '@timestamp': { order: 'desc' } }],
  };

  try {
    // เรียกผ่าน proxy แทน
    const response = await fetch(`${PROXY_URL}?index=${encodeURIComponent(esIndex)}`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify(requestBody),
    });

    if (!response.ok) {
      const errorText = await response.text();
      throw new Error(`Elasticsearch error: ${response.statusText} - ${errorText}`);
    }

    const data: ElasticsearchResponse = await response.json();
    
    // ตรวจสอบว่ามีข้อมูลหรือไม่
    if (!data.hits || !data.hits.hits || data.hits.hits.length === 0) {
      console.warn('No data returned from Elasticsearch');
      return [];
    }

    return data.hits.hits.map((hit) => ({
      id: hit._id,
      ...hit._source,
    }));
  } catch (error) {
    console.error('Error fetching MITRE techniques:', error);
    throw error;
  }
}
