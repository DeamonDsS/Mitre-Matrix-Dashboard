import React from "react";
import { useState, useEffect, useMemo } from 'react';
import { Shield, Activity, AlertTriangle, TrendingUp, Settings, RefreshCw } from 'lucide-react';
import TechniqueCard from "../../components/TechniqueCard";
import FilterBar from "../../components/FilterBar";
import StatsCard from "../../components/StatCard";
import ConfigModal from "../../components/ConfigModal";
import type { MitreTechnique, FilterState } from '../../types/mitre';
import { fetchMitreTechniques, getMockData } from '../../services/elasticsearch';
import { fetchMitreTechniques as fetchMitreTechniquesshell  } from "../../services/elasticpowershell";

const Dashboard: React.FC = () => {
  const [techniques, setTechniques] = useState<MitreTechnique[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [usingMockData, setUsingMockData] = useState(false);
  const [configModalOpen, setConfigModalOpen] = useState(false);
  const [esConfig, setEsConfig] = useState({
    esUrl: localStorage.getItem('esUrl') || 'http://localhost:9200',
    esIndex: localStorage.getItem('esIndex') || '.ds-winlogbeat-9.1.5-*',
  });

  const [filters, setFilters] = useState<FilterState>({
    search: '',
    tactic: 'all',
    severity: 'all',
  });

  const loadTechniques = async () => {
    setLoading(true);
    setError(null);

    try {
      if (esConfig.esUrl) {
        const data = await fetchMitreTechniques(esConfig.esUrl, esConfig.esIndex, filters);
        console.log("Show technique that get:", data);
        setTechniques(data);
        setUsingMockData(false);
      } else {
        throw new Error('No Elasticsearch URL configured');
      }
    } catch (err) {
      console.warn('Failed to fetch from Elasticsearch, using mock data:', err);
      setTechniques(getMockData());
      setUsingMockData(true);
      setError('Using demo data. Configure Elasticsearch to connect to your data.');
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    loadTechniques();
  }, [esConfig]);

  useEffect(() => {
    const timer = setTimeout(() => {
      if (esConfig.esUrl) {
        loadTechniques();
      }
    }, 500);

    return () => clearTimeout(timer);
  }, [filters]);

  const filteredTechniques = useMemo(() => {
    return techniques.filter((technique) => {
      const matchesSearch =
        !filters.search ||
        technique.technique_id.toLowerCase().includes(filters.search.toLowerCase()) ||
        technique.technique_name.toLowerCase().includes(filters.search.toLowerCase()) ||
        technique.description.toLowerCase().includes(filters.search.toLowerCase());

      const matchesTactic = filters.tactic === 'all' || technique.tactic === filters.tactic;

      const matchesSeverity =
        filters.severity === 'all' || technique.severity === filters.severity;

      return matchesSearch && matchesTactic && matchesSeverity;
    });
  }, [techniques, filters]);

  const stats = useMemo(() => {
    const severityCounts = filteredTechniques.reduce(
      (acc, t) => {
        acc[t.severity] = (acc[t.severity] || 0) + 1;
        return acc;
      },
      {} as Record<string, number>
    );

    const uniqueTactics = new Set(filteredTechniques.map((t) => t.tactic)).size;

    return {
      total: filteredTechniques.length,
      critical: severityCounts.critical || 0,
      high: severityCounts.high || 0,
      tactics: uniqueTactics,
    };
  }, [filteredTechniques]);

  const handleConfigSave = (config: { esUrl: string; esIndex: string }) => {
    localStorage.setItem('esUrl', config.esUrl);
    localStorage.setItem('esIndex', config.esIndex);
    setEsConfig(config);
  };

  return (
    <div className="min-h-screen bg-gray-50">
      <div className="max-w-7xl mx-auto px-4 py-8">
        <div className="flex flex-col md:flex-row md:items-center md:justify-between mb-8">
          <div className="flex items-center gap-3 mb-4 md:mb-0">
            <div className="w-12 h-12 bg-gray-900 rounded-lg flex items-center justify-center">
              <Shield className="w-7 h-7 text-white" />
            </div>
            <div>
              <h1 className="text-2xl md:text-3xl font-bold text-gray-900">
                MITRE ATT&CK Dashboard
              </h1>
              <p className="text-sm text-gray-600">
                Real-time threat technique monitoring
                {usingMockData && (
                  <span className="ml-2 text-orange-600 font-medium">(Demo Mode)</span>
                )}
              </p>
            </div>
          </div>

          <div className="flex gap-2">
            <button
              onClick={loadTechniques}
              disabled={loading}
              className="px-4 py-2 bg-white border border-gray-300 rounded-lg hover:bg-gray-50 transition-colors text-sm font-medium text-gray-700 flex items-center gap-2 disabled:opacity-50"
            >
              <RefreshCw className={`w-4 h-4 ${loading ? 'animate-spin' : ''}`} />
              Refresh
            </button>
            <button
              onClick={() => setConfigModalOpen(true)}
              className="px-4 py-2 bg-gray-900 text-white rounded-lg hover:bg-gray-800 transition-colors text-sm font-medium flex items-center gap-2"
            >
              <Settings className="w-4 h-4" />
              Configure
            </button>
          </div>
        </div>

        {error && (
          <div className="mb-6 bg-orange-50 border border-orange-200 rounded-lg p-4 flex items-start gap-3">
            <AlertTriangle className="w-5 h-5 text-orange-600 flex-shrink-0 mt-0.5" />
            <div>
              <p className="text-sm text-orange-800">{error}</p>
            </div>
          </div>
        )}

        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4 mb-6">
          <StatsCard
            title="Total Techniques"
            value={stats.total}
            icon={Activity}
            color="bg-gray-900"
          />
          <StatsCard
            title="Critical Severity"
            value={stats.critical}
            icon={AlertTriangle}
            color="bg-red-500"
          />
          <StatsCard title="High Severity" value={stats.high} icon={TrendingUp} color="bg-orange-500" />
          <StatsCard title="Unique Tactics" value={stats.tactics} icon={Shield} color="bg-blue-500" />
        </div>

        <FilterBar filters={filters} onFilterChange={setFilters} />

        {loading ? (
          <div className="flex items-center justify-center py-20">
            <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-gray-900"></div>
          </div>
        ) : filteredTechniques.length === 0 ? (
          <div className="text-center py-20">
            <Shield className="w-16 h-16 text-gray-300 mx-auto mb-4" />
            <h3 className="text-lg font-semibold text-gray-900 mb-2">No techniques found</h3>
            <p className="text-gray-600">Try adjusting your filters or search criteria</p>
          </div>
        ) : (
          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
            {filteredTechniques.map((technique) => (
              <TechniqueCard key={technique.id} technique={technique} />
            ))}
          </div>
        )}
      </div>

      <ConfigModal
        isOpen={configModalOpen}
        onClose={() => setConfigModalOpen(false)}
        onSave={handleConfigSave}
        currentConfig={esConfig}
      />
    </div>
  );
}

export default Dashboard;
