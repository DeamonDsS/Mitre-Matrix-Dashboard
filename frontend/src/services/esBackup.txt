import type { MitreTechnique, ElasticsearchResponse } from '../types/mitre';

interface EsQueryClause {
  multi_match?: {
    query: string;
    fields: string[];
    fuzziness: string;
  };
  match?: {
    [key: string]: string;
  };
  range?: {
    [key: string]: {
      gte?: string;
      lte?: string;
    };
  };
  terms?: {
    [key: string]: string[];
  };
}

export async function fetchMitreTechniques(
  esUrl: string,
  esIndex: string,
  filters?: {
    search?: string;
    tactic?: string;
    severity?: string;
  }
): Promise<MitreTechnique[]> {
  const query: { bool: { must: EsQueryClause[]; should?: EsQueryClause[] } } = {
    bool: {
      must: [],
    },
  };

  // Search across multiple fields
  if (filters?.search) {
    query.bool.must.push({
      multi_match: {
        query: filters.search,
        fields: ['event.code', 'message', 'winlog.event_data.*', 'host.name', 'user.name'],
        fuzziness: 'AUTO',
      },
    });
  }

  const requestBody = {
    query: query.bool.must.length > 0 ? query : { match_all: {} },
    size: 100,
    sort: [{ '@timestamp': { order: 'desc' } }],
  };

  try {
    const response = await fetch(`${PROXY_URL}?index=${encodeURIComponent(esIndex)}`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify(requestBody),
    });

    if (!response.ok) {
      const errorText = await response.text();
      throw new Error(`Elasticsearch error: ${response.statusText} - ${errorText}`);
    }

    const data: ElasticsearchResponse = await response.json();
    
    if (!data.hits || !data.hits.hits || data.hits.hits.length === 0) {
      console.warn('No data returned from Elasticsearch');
      return [];
    }

    console.log(`Found ${data.hits.hits.length} events from Elasticsearch`);
    
    // Log first event for debugging
    if (data.hits.hits.length > 0) {
      console.log('Sample event structure:', JSON.stringify(data.hits.hits[0]._source, null, 2));
    }

    // Transform Windows Events to MITRE-like format
    return data.hits.hits.map((hit) => {
      const source = hit._source || {};
      
      // Extract event code from different possible locations
      const eventCode = source.event?.code || 
                       source.winlog?.event_id?.toString() || 
                       'unknown';
      
      const eventInfo = EVENT_SEVERITY_MAP[eventCode] || { 
        severity: 'medium', 
        category: 'Unknown Event' 
      };
      
      // Extract user information
      const userName = source.user?.name || 
                      source.winlog?.event_data?.TargetUserName ||
                      source.winlog?.event_data?.SubjectUserName ||
                      source.related?.user?.[0] ||
                      'N/A';
      
      // Extract host information
      const hostName = source.host?.name || 
                      source.host?.hostname ||
                      source.agent?.name || 
                      source.winlog?.computer_name ||
                      'Unknown Host';
      
      // Extract process information
      const processName = source.process?.name || 
                         source.process?.executable ||
                         source.winlog?.event_data?.ProcessName ||
                         source.winlog?.event_data?.NewProcessName ||
                         '';
      
      // Extract channel
      const channel = source.winlog?.channel || 
                     source.event?.module || 
                     'Unknown';
      
      // Extract log level
      const logLevel = source.log?.level || 
                      source.winlog?.level ||
                      'info';
      
      // Build description from message
      let description = source.message || 'No description available';
      
      // Truncate long messages but keep meaningful content
      if (description.length > 250) {
        // Try to get first paragraph or first few sentences
        const firstParagraph = description.split('\n\n')[0];
        if (firstParagraph.length > 0 && firstParagraph.length <= 250) {
          description = firstParagraph;
        } else {
          description = description.substring(0, 250) + '...';
        }
      }
      
      // Extract platform
      const platform = source.host?.os?.family || 
                      source.host?.os?.platform || 
                      'Windows';

      return {
        id: hit._id,
        technique_id: `Event-${eventCode}`,
        technique_name: eventInfo.category,
        tactic: EVENT_TACTIC_MAP[eventCode] || 'Discovery',
        description: description,
        severity: eventInfo.severity as 'low' | 'medium' | 'high' | 'critical' | 'unknown',
        timestamp: source['@timestamp'] || new Date().toISOString(),
        platform: [platform],
        
        // Windows Event Log specific fields
        event_code: eventCode,
        host_name: hostName,
        user_name: userName,
        process_name: processName,
        log_level: logLevel,
        channel: channel,
        
        // Keep original source for debugging
        _raw: source,
      };
    }).filter((event) => {
      // Apply filters
      const matchesTactic = !filters?.tactic || 
                           filters.tactic === 'all' || 
                           event.tactic === filters.tactic;
      
      const matchesSeverity = !filters?.severity || 
                             filters.severity === 'all' || 
                             event.severity === filters.severity;
      
      return matchesTactic && matchesSeverity;
    });
  } catch (error) {
    console.error('Error fetching Windows Events:', error);
    throw error;
  }
}
